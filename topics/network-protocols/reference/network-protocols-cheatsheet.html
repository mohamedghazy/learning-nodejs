TCP vs UDP

TCP — node:net

  • Connection-oriented (3-way handshake)
  • Reliable delivery — retransmits lost packets
  • In-order delivery — reorders segments
  • Flow control + congestion control
  • Byte stream — no message boundaries
  • Higher latency, lower throughput ceiling
  • Used by: HTTP, SSH, DB connections, WebSocket

UDP — node:dgram

  • Connectionless — no handshake
  • Unreliable — no retransmit
  • No ordering guarantee
  • No flow/congestion control
  • Datagram — preserves message boundaries
  • Lower latency, higher throughput ceiling
  • Used by: DNS, video/gaming, WebRTC, QUIC

node:net Quick Reference

import net from 'node:net';

// Server
const server = net.createServer((socket) => { /* Duplex stream */ });
server.listen(port, host, callback);
server.close(callback);

// Socket events
socket.on('data', (chunk) => {});       // Buffer received
socket.on('end', () => {});             // Peer sent FIN
socket.on('close', (hadError) => {});  // Socket fully closed
socket.on('error', (err) => {});      // ECONNRESET etc.
socket.on('drain', () => {});         // Write buffer empty again

// Socket methods
socket.write(data);          // Returns false if buffer full (backpressure)
socket.end([data]);          // Send FIN after optional data
socket.destroy([err]);      // Forcibly close (RST)
socket.setEncoding('utf8'); // Decode chunks as strings
socket.setTimeout(ms);      // Emit 'timeout' after ms of inactivity

// Client
const client = net.createConnection({ port, host }, connectCallback);

node:dgram Quick Reference

import dgram from 'node:dgram';

// Create socket
const sock = dgram.createSocket('udp4');  // or 'udp6'

// Bind (receive)
sock.bind(port, address);

// Events
sock.on('message', (msg, rinfo) => {});
// msg: Buffer, rinfo: { address, family, port, size }
sock.on('listening', () => {});
sock.on('error', (err) => {});
sock.on('close', () => {});

// Send
sock.send(buf, offset, length, port, address, callback);
sock.send(buf, port, address);             // Short form

// Broadcast
sock.setBroadcast(true);
sock.send(buf, port, '255.255.255.255');

// Multicast
sock.addMembership('224.1.1.1');
sock.dropMembership('224.1.1.1');
sock.setMulticastTTL(128);               // Hops before packet is dropped

TLS Quick Reference

import tls from 'node:tls';

// Server
const server = tls.createServer({
  key:  fs.readFileSync('server.key'),
  cert: fs.readFileSync('server.crt'),
  minVersion: 'TLSv1.2',
  // For mTLS:
  requestCert: true, rejectUnauthorized: true,
  ca: fs.readFileSync('client-ca.crt'),
}, handler);

// Client
const sock = tls.connect({ host, port,
  // NEVER set this false in production:
  rejectUnauthorized: true,
}, callback);
