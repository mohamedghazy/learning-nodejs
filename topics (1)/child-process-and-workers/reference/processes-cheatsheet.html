child_process Methods

import { exec, execFile, spawn, fork } from 'node:child_process';

// exec: shell command, buffered output (< 1MB)
exec('git status', (err, stdout, stderr) => {});

// spawn: streaming output, no shell by default
const child = spawn('ffmpeg', ['-i', 'input.mp4', 'output.webm']);
child.stdout.on('data', chunk => {});

// fork: Node.js script with IPC channel
const worker = fork('./task.js');
worker.send({ data }); worker.on('message', result => {});

Worker Threads

// main.js
import { Worker } from 'node:worker_threads';
const w = new Worker('./task.js', { workerData: { input } });
w.on('message', result => {});
w.on('error', err => {});

// task.js
import { parentPort, workerData } from 'node:worker_threads';
const result = compute(workerData);
parentPort.postMessage(result);

Cluster

import cluster from 'node:cluster';
import os from 'node:os';

if (cluster.isPrimary) {
  for (let i = 0; i < os.cpus().length; i++) cluster.fork();
  cluster.on('exit', () => cluster.fork()); // auto-restart
} else {
  // Each worker runs the server
  http.createServer(handler).listen(3000);
}
